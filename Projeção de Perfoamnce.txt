select REN.CLCLI_CD, REN.DT_POS, REN.PC_MES_CT, REN.PC_MES_I1  
,PC_ANO_CT
,PC_ANO_I1
,PC_12M_CT
,PC_12M_I1

, DI_A.TAXA, DI_A.DU
, DI_B.TAXA, DI_B.DU

, VAR.VAR
, VAR.VAR21
, VAR.ES
, VAR.ESTRESSE
, PL.VL_PATRLIQTOT1

from dl_Gestao_recursos.sac_cl_rentab REN
left join (SELECT DATA, CONTRATO, DU, TAXA, LAG(DU) OVER (PARTITION BY DATA ORDER BY DU) Dlag FROM DL_GESTAO_RECURSOS.CURVA_DI
) DI_A ON DI_A.DATA = REN.DT_POS AND DI_A.DLAG IS NULL
LEFT JOIN DL_GESTAO_RECURSOS.CURVA_DI DI_B ON DI_B.DATA = REN.DT_POS AND DI_B.CONTRATO = 'DI1F26'
LEFT JOIN DL_GESTAO_RECURSOS.VAR_ESTRATEGIA_SAS VAR ON VAR.DTPOS = REN.DT_POS AND VAR.NOMECARTEIRA = REN.CLCLI_CD AND VAR.GRUPO = 'Total'
LEFT JOIN DL_GESTAO_RECURSOS.SAC_CL_PATRSINT PL ON PL.DT = REN.DT_POS AND PL.CLCLI_CD = REN.CLCLI_CD
where REN.dt_pos = '2024-12-19'

--WHERE DATA = '2024-12-27'
